@triton.jit
def baseline(A, B, C):
    rows = tl.arange(0, 32)
    cols = tl.arange(0, 32)
    a = tl.load(A + rows[:, None] * 32 + cols[None, :])
    b = tl.load(B + rows[:, None] * 32 + cols[None, :])
    acc = tl.dot(a, b, tl.full((32, 32), 0.0, tl.float32))
    acc = tl.maximum(acc, 0.0)
    tl.store(C + rows[:, None] * 32 + cols[None, :], acc)
